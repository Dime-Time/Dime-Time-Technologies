from pptx import Presentation

from pptx.enum.text import PP_ALIGN

# ─────────────────────────────────────────────
# BRAND
# ─────────────────────────────────────────────

prs = Presentation()

# ─────────────────────────────────────────────
# HELPERS
# ─────────────────────────────────────────────
def set_background(slide, color):
    bg = slide.background
    fill = bg.fill
    fill.solid()
    fill.fore_color.rgb = color

def add_title(slide, text, color=WHITE):
    txBox = slide.shapes.add_textbox(Inches(1), Inches(1.2), Inches(8), Inches(1.5))
    tf = txBox.text_frame
    p = tf.paragraphs[0]
    run = p.add_run()
    run.text = text
    run.font.size = Pt(44)
    run.font.bold = True
    run.font.color.rgb = color
    p.alignment = PP_ALIGN.CENTER

def add_subtitle(slide, text, color=WHITE):
    txBox = slide.shapes.add_textbox(Inches(1.5), Inches(2.6), Inches(7), Inches(1.5))
    tf = txBox.text_frame
    p = tf.paragraphs[0]
    run = p.add_run()
    run.text = text
    run.font.size = Pt(20)
    run.font.color.rgb = color
    p.alignment = PP_ALIGN.CENTER

def add_bullets(slide, bullets, top=2.5, color=BLACK):
    txBox = slide.shapes.add_textbox(Inches(1), Inches(top), Inches(8), Inches(4))
    tf = txBox.text_frame
    tf.clear()
    for i, b in enumerate(bullets):
        p = tf.add_paragraph() if i > 0 else tf.paragraphs[0]
        p.text = b
        p.font.size = Pt(22)
        p.font.color.rgb = color
        p.level = 0

def add_section_title(slide, title):
    txBox = slide.shapes.add_textbox(Inches(1), Inches(0.8), Inches(8), Inches(1))
    tf = txBox.text_frame
    p = tf.paragraphs[0]
    run = p.add_run()
    run.text = title
    run.font.size = Pt(34)
    run.font.bold = True
    run.font.color.rgb = BLACK

def add_allocation_diagram(slide):
    # Center node
    center = slide.shapes.add_shape(MSO_SHAPE.OVAL, Inches(4), Inches(2.5), Inches(1.2), Inches(1.2))
    center.fill.solid()
    center.fill.fore_color.rgb = PRIMARY
    center.text_frame.text = "Round-Ups"

    # Left (Debt)
    left = slide.shapes.add_shape(MSO_SHAPE.RECTANGLE, Inches(1.5), Inches(2.6), Inches(2), Inches(1))
    left.fill.solid()
    left.fill.fore_color.rgb = RGBColor(230, 230, 255)
    left.text_frame.text = "Debt Repayment"

    # Right (Crypto)
    right = slide.shapes.add_shape(MSO_SHAPE.RECTANGLE, Inches(6.5), Inches(2.6), Inches(2), Inches(1))
    right.fill.solid()
    right.fill.fore_color.rgb = RGBColor(230, 230, 255)
    right.text_frame.text = "Bitcoin"
